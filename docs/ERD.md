# Clubwire ERD

> 자동 생성: `python scripts/gen_docs.py`. 영역별로 나눠 그렸고, 다른 영역 테이블은 키만 표시.

## 1. Club / Member / Config

```mermaid
erDiagram
    CLUB {
        BIGINT CLUB_ID PK
        VARCHAR CLUB_SLUG
        VARCHAR CLUB_NAME
        VARCHAR SPORT_TYPE
        VARCHAR TIMEZONE
        INTEGER CONFIG_VERSION
        BOOL ACTIVE_YN
        DATETIME UPDATED_AT
        DATETIME CREATED_AT
    }
    CLUB_ADMIN {
        BIGINT ADMIN_ID PK
        BIGINT CLUB_ID FK
        VARCHAR ADMIN_NAME
        VARCHAR PASSWORD_HASH
        BOOL ACTIVE_YN
        DATETIME LAST_LOGIN_AT
        DATETIME UPDATED_AT
        DATETIME CREATED_AT
    }
    MEMBER {
        BIGINT MEMBER_ID PK
        BIGINT CLUB_ID FK
        VARCHAR NAME
        VARCHAR PROFILE_IMAGE_URL
        INTEGER DISPLAY_ORDER
        BOOL ACTIVE_YN
        DATE JOINED_AT
        DOUBLE INITIAL_RATING
        DATETIME UPDATED_AT
        DATETIME CREATED_AT
    }
    SYSTEM_CONFIG {
        BIGINT CLUB_ID PK
        VARCHAR CONFIG_KEY PK
        VARCHAR CONFIG_VALUE
        VARCHAR VALUE_TYPE
        VARCHAR DESCRIPTION
        DATETIME UPDATED_AT
    }
    SYSTEM_CONFIG_HISTORY {
        BIGINT HISTORY_ID PK
        BIGINT CLUB_ID FK
        VARCHAR CONFIG_KEY
        VARCHAR BEFORE_VALUE
        VARCHAR AFTER_VALUE
        INTEGER CONFIG_VERSION
        BIGINT CHANGED_BY_ADMIN_ID FK
        DATETIME CHANGED_AT
    }
    CLUB ||--o{ CLUB_ADMIN : "CLUB_ID"
    CLUB ||--o{ MEMBER : "CLUB_ID"
    CLUB ||--o{ SYSTEM_CONFIG : "CLUB_ID"
    CLUB ||--o{ SYSTEM_CONFIG_HISTORY : "CLUB_ID"
    CLUB_ADMIN |o--o{ SYSTEM_CONFIG_HISTORY : "CHANGED_BY_ADMIN_ID"
```
